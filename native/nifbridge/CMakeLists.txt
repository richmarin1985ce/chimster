cmake_minimum_required(VERSION 3.16)
project(chimster_nifbridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(NIFLY_SOURCE "" CACHE PATH "Pinned nifly checkout")
if(NOT EXISTS "${NIFLY_SOURCE}/include/NifFile.hpp")
  message(FATAL_ERROR "Supply -DNIFLY_SOURCE=/path/to/nifly")
endif()
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
add_subdirectory("${NIFLY_SOURCE}" nifly-build EXCLUDE_FROM_ALL)
add_executable(chimster-nifbridge main.cpp)
target_link_libraries(chimster-nifbridge PRIVATE nifly)
